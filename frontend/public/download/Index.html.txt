// Get tanggal hari ini di timezone LOKAL user (bukan UTC), format YYYY-MM-DD
function getTodayLocal() {
    const d = new Date();
    const yyyy = d.getFullYear();
    const mm = String(d.getMonth() + 1).padStart(2, '0');
    const dd = String(d.getDate()).padStart(2, '0');
    return yyyy + '-' + mm + '-' + dd;
}

// Set tanggal default tiap kali pages dibuka (bukan sekali saat load)
function setTanggalHariIni() {
    const today = getTodayLocal();
    ['inputAbsenTanggal', 'inputNilaiTanggal', 'inputJurnalTanggal'].forEach(id => {
        const inp = document.getElementById(id);
        if (inp && !inp.value) inp.value = today;
    });
}

// === HELPER: aman buat element dgn text (no XSS) ===
function el(tag, opts) {
    const e = document.createElement(tag);
    if (!opts) return e;
    if (opts.text != null) e.textContent = opts.text;
    if (opts.cls) e.className = opts.cls;
    if (opts.html != null) e.innerHTML = opts.html; // gunakan ini hanya untuk konten yg tidak dari user
    if (opts.attrs) for (const k in opts.attrs) e.setAttribute(k, opts.attrs[k]);
    if (opts.style) e.style.cssText = opts.style;
    if (opts.onclick) e.onclick = opts.onclick;
    return e;
}

function prosesLogin() {
    clearLoginError();
    const u = document.getElementById('loginUsername').value;
    const p = document.getElementById('loginPassword').value;
    if (!u || !p) {
        showLoginError('Username dan password wajib diisi.');
        return;
    }
    setLoginLoading(true);
    google.script.run.withSuccessHandler(function(res) {
        setLoginLoading(false);
        if (res.success) {
            applyUserSession(res.user);
            document.getElementById('loginOverlay').style.display = 'none';
            document.getElementById('sidebar').style.display = 'block';
            document.getElementById('mainContent').style.display = 'block';
            if (res.initialData) onDataLoaded(res.initialData);
            else refreshSemuaData();
            if (res.dashboardStats) renderDashboardStats(res.dashboardStats);
            else loadDashboardStats();
            checkPendingAbsensi(); // cek antrian offline
            initRekapTahun();
            initPengampuModule();
            showToast(res.message, 'success');
        } else {
            showLoginError(res.message || 'Username atau Password salah.');
            // Fokuskan ke field password supaya user tinggal ketik ulang
            document.getElementById('loginPassword').focus();
            document.getElementById('loginPassword').select();
        }
    }).withFailureHandler(function(err) {
        setLoginLoading(false);
        showLoginError('Gagal menghubungi server: ' + (err.message || err));
    }).cekLoginCepat(u, p);
}

function onDataLoaded(data) {
    toggleLoading(false);
    if (data.appInfo) applyAppInfo(data.appInfo);
    // Filter kelas untuk pengampu
    let allKelas = data.kelas || [];
    if (currentUser && String(currentUser.role).toLowerCase() !== 'admin') {
        const allowed = new Set((currentUser.kelasDiampu || []).map(k => String(k||'').trim()));
        allKelas = allKelas.filter(k => allowed.has(String((Array.isArray(k)?k[1]:k.nama)||'').trim()));
    }
    globalDataKelas = allKelas;
    globalDataMapel = data.mapel || [];
    updateDropdownKelas();
    updateDropdownMapel();
    renderTableKelas();
    renderTableMapel();
    siswaLoaded = false;
    jadwalLoaded = false;
    if (document.getElementById('siswa').classList.contains('active')) loadSiswaData();
    if (document.getElementById('jadwal').classList.contains('active')) loadJadwalData();
    if (data.setting && data.setting.length > 0) {
        const s = data.setting[0];
        document.getElementById('inputSetSekolah').value = s[0] || '';
        document.getElementById('inputSetAlamat').value = s[1] || '';
        document.getElementById('inputSetGuru').value = s[2] || '';
        document.getElementById('inputSetNipGuru').value = s[3] || '';
        document.getElementById('inputSetKepsek').value = s[4] || '';
        document.getElementById('inputSetNipKepsek').value = s[5] || '';
        if (document.getElementById('inputSetLogo')) document.getElementById('inputSetLogo').value = s[6] || '';
        applyAppInfo({
            appName: s[0] || currentAppName,
            namaSekolah: s[0] || '',
            alamat: s[1] || '',
            namaGuru: s[2] || '',
            nipGuru: s[3] || '',
            namaKepsek: s[4] || '',
            nipKepsek: s[5] || '',
            logoUrl: s[6] || ''
        });
        if (currentUser && currentUser.nama) {
            document.getElementById('dashboardGreeting').innerText = 'Halo, ' + currentUser.nama + '!';
        } else if (s[2]) {
            document.getElementById('dashboardGreeting').innerText = 'Halo, ' + s[2] + '!';
        }

    if (typeof initDashboardKehadiran === 'function') initDashboardKehadiran();
    if (typeof loadDashPeringatan === 'function') loadDashPeringatan();
}

function updateDropdownKelas() {
    const selects = document.querySelectorAll('.dropdown-kelas');
    selects.forEach(sel => {
        const prev = sel.value;
        const withSemua = (sel.id === 'filterSiswaKelas' || sel.id === 'inputRekapKelas');
        let placeholder, placeholderValue;
        if (withSemua) { placeholder = 'Semua Kelas'; placeholderValue = 'Semua'; }
        else { placeholder = '-- Pilih Kelas --'; placeholderValue = ''; }
        const opts = [el('option', { text: placeholder, attrs: { value: placeholderValue } })];
        globalDataKelas.forEach(k => {
            opts.push(el('option', { text: k[1], attrs: { value: k[1] } }));
        });
        sel.innerHTML = '';
        opts.forEach(o => sel.appendChild(o));
        if (prev) sel.value = prev;
    });
}

    });
}

function renderTableKelas() {
    const tbody = document.getElementById('tableBodyKelas');
    tbody.innerHTML = '';
    if (globalDataKelas.length === 0) {
        tbody.appendChild(el('tr', { html: '<td colspan="3" class="text-muted">Belum ada kelas.</td>' }));
        return;
    }
    globalDataKelas.forEach(function(row, i) {
        const tr = el('tr');
        tr.appendChild(el('td', { text: i + 1 }));
        tr.appendChild(el('td', { text: row[1] }));
        const tdAksi = el('td');
        const btnEdit = el('button', { cls: 'btn btn-sm btn-warning btn-action', html: '<i class="fas fa-edit"></i>' });
        btnEdit.onclick = function() { bukaModalEditKelas(row[0], row[1]); };

        btnHapus.onclick = function() { konfirmHapus('Kelas', row[0], row[1]); };
        tdAksi.appendChild(btnEdit);
        tdAksi.appendChild(btnHapus);
        tr.appendChild(tdAksi);
        tbody.appendChild(tr);
    });
}

        tr.appendChild(el('td', { text: row[1] }));

let absensiState = {};

// Track apakah saat ini mode edit (ada data existing)
let absenIsEditMode = false;

function loadFormAbsensi() {
    const t = document.getElementById('inputAbsenTanggal').value;
    const kls = document.getElementById('inputAbsenKelas').value;
    const mapel = document.getElementById('inputAbsenMapel').value;
    if (!t) return showToast('Pilih tanggal dulu!', 'warning');
    if (!kls) return showToast('Pilih kelas dulu!', 'warning');
    if (!mapel) return showToast('Pilih mata pelajaran dulu!', 'warning');

    toggleLoading(true, 'Memuat siswa & cek data...');

    // Fetch parallel: getSiswaByKelas + cekAbsensiTanggal
    let siswaResult = null, cekResult = null, errorOccurred = false;
    let pending = 2;

    function tryFinish() {
        if (--pending > 0) return;
        toggleLoading(false);
        if (errorOccurred) {
            // Coba cache offline
            const cached = getCachedSiswa(kls);
            if (cached && cached.length > 0) {
                setOfflineMode(true);
                showToast('Mode offline: pakai data tersimpan (cek data existing tidak tersedia)', 'warning');
                handleSiswaAbsensi(cached, kls, { exists: false, data: {} });
            } else {
                showToast('Gagal muat siswa & tidak ada cache.', 'error');
            }
            return;
        }
        handleSiswaAbsensi(siswaResult, kls, cekResult || { exists: false, data: {} });
    }

    google.script.run
        .withSuccessHandler(function(s) { siswaResult = s; tryFinish(); })
        .withFailureHandler(function() { errorOccurred = true; tryFinish(); })
        .getSiswaByKelas(kls);

    google.script.run
        .withSuccessHandler(function(c) { cekResult = c; tryFinish(); })
        .withFailureHandler(function() { cekResult = { exists: false, data: {} }; tryFinish(); })
        .cekAbsensiTanggal(t, kls, mapel);
}

function handleSiswaAbsensi(s, kls, existingData) {
    if (!s || s.length === 0) return showToast('Tidak ada siswa di kelas ini', 'error');
    absensiState = {};
    absenSiswaList = s;
    btnRefs = new Map();

    // Pre-set status: pakai data existing kalau ada, default Hadir
    const exData = (existingData && existingData.data) ? existingData.data : {};
    s.forEach((siswa, i) => {
        const noUrut = siswa[1]; // [ID, No, Nama, Kelas]
        absensiState[i] = exData[noUrut] || 'Hadir';
    });

    // Tampilkan badge mode edit
    absenIsEditMode = !!(existingData && existingData.exists);
    const badge = document.getElementById('absenModeEdit');
    const btnText = document.getElementById('btnKirimAbsenText');
    const btn = document.getElementById('btnKirimAbsen');
    if (absenIsEditMode) {
        badge.style.display = 'block';
        document.getElementById('absenModeEditMsg').innerText =
            'Sudah ada absensi (' + existingData.count + ' siswa) untuk tanggal, kelas, dan mapel ini. Status di bawah sudah dipre-load. Klik "Update Absensi" untuk menyimpan perubahan.';
        btnText.innerText = 'Update Absensi';
        btn.classList.remove('btn-success');
        btn.classList.add('btn-warning');
    } else {
        badge.style.display = 'none';
        btnText.innerText = 'Kirim Absensi';
        btn.classList.remove('btn-warning');
        btn.classList.add('btn-success');
    }

    const container = document.getElementById('areaAbsensiSiswa');
    container.innerHTML = '';
    document.getElementById('legendaAbsen').style.display = 'flex';
    document.getElementById('formAbsensiFooter').style.display = 'none';
    renderAbsensiChunk(s, 0, 10, container);
}

function renderAbsensiChunk(siswa, start, chunkSize, container) {
    const end = Math.min(start + chunkSize, siswa.length);
    const fragment = document.createDocumentFragment();
    for (let i = start; i < end; i++) {
        const x = siswa[i];
        const noVal = x[1];
        const namaVal = x[2];
        const currentStatus = absensiState[i] || 'Hadir';
        const row = el('div', { cls: 'absen-row' });
        row.appendChild(el('span', { cls: 'absen-no', text: noVal }));
        row.appendChild(el('span', { cls: 'absen-nama', text: namaVal }));
        const btnsDiv = el('div', { cls: 'absen-btns' });
        const rowBtns = [];
        STATUS_LIST.forEach(st => {
            const btn = el('button', {
                cls: 'absen-btn' + (st === currentStatus ? ' ' + AKTIF_CLASS[st] : ''),
                text: STATUS_KODE[st],
                attrs: { type: 'button', 'data-idx': i, 'data-status': st }
            });
            rowBtns.push(btn);
            btnsDiv.appendChild(btn);
        });
        btnRefs.set(i, rowBtns);
        row.appendChild(btnsDiv);
        fragment.appendChild(row);
    }
    container.appendChild(fragment);
    if (end < siswa.length) {
        setTimeout(() => renderAbsensiChunk(siswa, end, chunkSize, container), 16);
    } else {
        document.getElementById('formAbsensiFooter').style.display = 'block';
    }
}

function kirimAbsensi() {
    const t = document.getElementById('inputAbsenTanggal').value;
    const k = document.getElementById('inputAbsenKelas').value;
    const m = document.getElementById('inputAbsenMapel').value;
    if (!t || !k || !m) return showToast('Lengkapi tanggal, kelas, dan mata pelajaran!', 'warning');
    if (absenSiswaList.length === 0) return showToast('Belum ada data absensi', 'warning');

    // Konfirmasi kalau mode edit (akan replace data lama)
    if (absenIsEditMode) {
        if (!confirm('Update absensi untuk tanggal ' + t + ' kelas ' + k + ' mapel ' + m + '? Data lama akan diganti.')) return;
    }

    // Format payload: array of {no, nama, status}
    const dataSiswa = absenSiswaList.map((x, i) => ({
        no: x[1], nama: x[2], status: absensiState[i] || 'Hadir'
    }));

    toggleLoading(true, absenIsEditMode ? 'Memperbarui absensi...' : 'Menyimpan absensi...');
    google.script.run
        .withSuccessHandler(function(r) {
            toggleLoading(false);
            showToast(r);
            resetAbsensiUI();
        })
        .withFailureHandler(function(err) {
            toggleLoading(false);
            // Simpan ke localStorage untuk sync nanti
            const queue = getPendingAbsensi();
            queue.push({ tanggal: t, kelas: k, mapel: m, dataSiswa: dataSiswa, ts: Date.now() });
            savePendingAbsensi(queue);
            setOfflineMode(true);
            showToast('Tersimpan offline. Akan otomatis dikirim saat online.', 'warning');
            resetAbsensiUI();
            checkPendingAbsensi();
        })
        .simpanAbsensiMasal(t, k, m, dataSiswa, _actor());
}

    document.getElementById('formAbsensiFooter').style.display = 'none';
    document.getElementById('absenModeEdit').style.display = 'none';
    absensiState = {}; absenSiswaList = []; btnRefs = new Map();
    absenIsEditMode = false;
}

function cacheSiswaForOffline() {
    try {
        const cache = {};
        globalDataSiswa.forEach(s => {
            const kls = s[3];
            if (!cache[kls]) cache[kls] = [];
            cache[kls].push(s);
        });
        localStorage.setItem('myadteach_siswa_cache', JSON.stringify(cache));
    } catch (e) { /* ignore quota */ }
}

function getCachedSiswa(kelas) {
    try {
        const c = JSON.parse(localStorage.getItem('myadteach_siswa_cache') || '{}');
        return c[kelas] || [];
    } catch (e) { return []; }
}

function getPendingAbsensi() {
    try { return JSON.parse(localStorage.getItem('myadteach_pending_absensi') || '[]'); }
    catch (e) { return []; }
}

function savePendingAbsensi(arr) {
    try { localStorage.setItem('myadteach_pending_absensi', JSON.stringify(arr)); }
    catch (e) { showToast('Gagal simpan offline (storage penuh)', 'error'); }
}

function setOfflineMode(isOffline) {
    document.getElementById('offlineBadge').classList.toggle('show', isOffline);
}

function checkPendingAbsensi() {
    const queue = getPendingAbsensi();
    const area = document.getElementById('pendingAbsenArea');
    area.innerHTML = '';
    if (queue.length === 0) return;
    const div = el('div', { cls: 'pending-sync' });
    div.appendChild(el('span', {
        html: '<i class="fas fa-clock text-warning"></i> <strong>' + queue.length + '</strong> absensi tertunda (offline). '
    }));
    const btn = el('button', { cls: 'btn btn-sm btn-warning ms-2', text: 'Kirim Sekarang' });
    btn.onclick = syncPendingAbsensi;
    div.appendChild(btn);
    area.appendChild(div);
}

function syncPendingAbsensi() {
    const queue = getPendingAbsensi();
    if (queue.length === 0) return;
    
    // PERINGATAN: kalau ada item di queue dan ada data online lebih baru yang sama tanggal+kelas,
    // sync akan overwrite. Beritahu user.
    const summary = queue.map((q, i) => 
        (i + 1) + '. Tanggal ' + q.tanggal + ' kelas ' + q.kelas + ' mapel ' + (q.mapel || '-') + ' (' + q.dataSiswa.length + ' siswa)'
    ).join('\n');
    
    if (!confirm(
        'Akan mengirim ' + queue.length + ' absensi offline:\n\n' + summary + 
        '\n\nPERHATIAN: Jika selama offline ada data absensi yang sudah di-input untuk tanggal+kelas+mapel yang sama, data tersebut akan diganti dengan data offline ini. Lanjutkan?'
    )) {
        return;
    }
    
    toggleLoading(true, 'Sinkronisasi ' + queue.length + ' absensi...');

    let successCount = 0;
    let failedQueue = [];

    function processNext(idx) {
        if (idx >= queue.length) {
            toggleLoading(false);
            savePendingAbsensi(failedQueue);
            if (failedQueue.length === 0) {
                setOfflineMode(false);
                showToast('Semua ' + successCount + ' absensi berhasil dikirim!', 'success');
            } else {
                showToast(successCount + ' sukses, ' + failedQueue.length + ' masih tertunda.', 'warning');
            }
            checkPendingAbsensi();
            return;
        }
        const item = queue[idx];
        // Validasi sebelum kirim — kalau struktur queue rusak, skip
        if (!item || !item.tanggal || !item.kelas || !item.mapel || !Array.isArray(item.dataSiswa)) {
            console.warn('Queue item rusak di index ' + idx + ', skipping:', item);
            processNext(idx + 1);
            return;
        }
        google.script.run
            .withSuccessHandler(function() {
                successCount++;
                processNext(idx + 1);
            })
            .withFailureHandler(function() {
                failedQueue.push(item);
                processNext(idx + 1);
            })
            .simpanAbsensiMasal(item.tanggal, item.kelas, item.mapel, item.dataSiswa, _actor());
    }
    processNext(0);
}

let nilaiExistingData = {}; // map noUrut → { nilai, bab, tujuan, bentuk }

function loadFormNilai() {
    const t = document.getElementById('inputNilaiTanggal').value;
    const kls = document.getElementById('inputNilaiKelas').value;
    const mapel = document.getElementById('inputNilaiMapel').value;
    const kat = document.getElementById('inputNilaiKategori').value;
    if (!t) return showToast('Pilih tanggal dulu!', 'warning');
    if (!kls) return showToast('Pilih kelas dulu!', 'warning');
    if (!mapel) return showToast('Pilih mata pelajaran!', 'warning');
    if (!kat) return showToast('Pilih kategori!', 'warning');

    toggleLoading(true, 'Memuat siswa & cek nilai...');

    let siswaResult = null, cekResult = null, errorOccurred = false;
    let pending = 2;

    function tryFinish() {
        if (--pending > 0) return;
        toggleLoading(false);
        if (errorOccurred || !siswaResult) {
            showToast('Gagal memuat data siswa', 'error');
            return;
        }
        if (siswaResult.length === 0) return showToast('Tidak ada siswa di kelas ini', 'error');
        handleNilaiSiswa(siswaResult, cekResult || { exists: false, data: {}, meta: { bab: '', tujuan: '', bentuk: '' } });
    }

    google.script.run
        .withSuccessHandler(function(s) { siswaResult = s; tryFinish(); })
        .withFailureHandler(function() { errorOccurred = true; tryFinish(); })
        .getSiswaByKelas(kls);

    google.script.run
        .withSuccessHandler(function(c) { cekResult = c; tryFinish(); })
        .withFailureHandler(function() { cekResult = { exists: false, data: {}, meta: { bab: '', tujuan: '', bentuk: '' } }; tryFinish(); })
        .cekNilaiTanggal(t, kls, mapel, kat);
}

function handleNilaiSiswa(s, existingData) {
    nilaiSiswaList = s;
    nilaiInputRefs = new Map();
    nilaiExistingData = (existingData && existingData.data) ? existingData.data : {};
    nilaiIsEditMode = !!(existingData && existingData.exists);

    // Pre-fill meta (bab/tujuan/bentuk) jika ada
    if (nilaiIsEditMode && existingData.meta) {
        document.getElementById('inputNilaiBab').value = existingData.meta.bab || '';
        document.getElementById('inputNilaiTujuan').value = existingData.meta.tujuan || '';
        document.getElementById('inputNilaiBentuk').value = existingData.meta.bentuk || '';
    } else {
        ['inputNilaiBab','inputNilaiTujuan','inputNilaiBentuk'].forEach(id => document.getElementById(id).value = '');
    }

    // Tampilkan badge & ubah tombol
    const badge = document.getElementById('nilaiModeEdit');
    const btnText = document.getElementById('btnKirimNilaiText');
    const btn = document.getElementById('btnKirimNilai');
    if (nilaiIsEditMode) {
        badge.style.display = 'block';
        document.getElementById('nilaiModeEditMsg').innerText =
            'Sudah ada nilai (' + existingData.count + ' siswa) untuk kombinasi tanggal/kelas/mapel/kategori ini. Nilai sudah dipre-fill. Edit yang perlu, lalu klik "Update Nilai".';
        btnText.innerText = 'Update Nilai';
        btn.classList.remove('btn-success');
        btn.classList.add('btn-warning');
    } else {
        badge.style.display = 'none';
        btnText.innerText = 'Kirim Nilai';
        btn.classList.remove('btn-warning');
        btn.classList.add('btn-success');
    }

    const tbody = document.getElementById('areaNilaiSiswa');
    tbody.innerHTML = '';
    renderNilaiChunk(s, 0, 15, tbody);
}

function renderNilaiChunk(siswa, start, chunkSize, tbody) {
    const end = Math.min(start + chunkSize, siswa.length);
    const fragment = document.createDocumentFragment();
    for (let i = start; i < end; i++) {
        const x = siswa[i];
        const noUrut = x[1];
        const existingNilai = nilaiExistingData[noUrut];
        const tr = el('tr');
        if (existingNilai) tr.style.backgroundColor = '#fffaf0';

        tr.appendChild(el('td', { text: noUrut, cls: 'text-center' }));

        const tdNama = el('td', { cls: 'text-start', style: 'overflow:hidden;text-overflow:ellipsis;white-space:nowrap;' });
        tdNama.appendChild(document.createTextNode(x[2]));
        if (existingNilai) {
            tdNama.appendChild(document.createTextNode(' '));
            tdNama.appendChild(el('span', { cls: 'badge bg-success', text: 'Tersimpan', style: 'font-size:0.65rem;' }));
        }
        tr.appendChild(tdNama);

        const tdNilai = el('td');
        const input = el('input', {
            cls: 'form-control form-control-sm text-center',
            attrs: { type: 'number', min: 0, max: 100, step: '0.01', inputmode: 'decimal', 'data-idx': i, placeholder: '0-100' }
        });
        if (existingNilai) input.value = existingNilai.nilai;
        tdNilai.appendChild(input);
        nilaiInputRefs.set(i, input);
        tr.appendChild(tdNilai);
        fragment.appendChild(tr);
    }
    tbody.appendChild(fragment);
    if (end < siswa.length) {
        setTimeout(() => renderNilaiChunk(siswa, end, chunkSize, tbody), 16);
    } else {
        document.getElementById('formNilai').style.display = 'block';
    }
}

document.getElementById('formNilai').onsubmit = function(e) {
    e.preventDefault();
    if (nilaiSiswaList.length === 0) return showToast('Muat data siswa dulu', 'warning');
    const t = document.getElementById('inputNilaiTanggal').value;
    const k = document.getElementById('inputNilaiKelas').value;
    const mp = document.getElementById('inputNilaiMapel').value;
    const kat = document.getElementById('inputNilaiKategori').value;
    const bab = document.getElementById('inputNilaiBab').value;
    const tujuan = document.getElementById('inputNilaiTujuan').value;
    const bentuk = document.getElementById('inputNilaiBentuk').value;

    const dataSiswa = nilaiSiswaList.map((x, i) => {
        const inp = nilaiInputRefs.get(i);
        return { no: x[1], nama: x[2], nilai: inp ? inp.value : '' };
    });
    const isiNilaiCount = dataSiswa.filter(d => d.nilai !== '').length;
    if (isiNilaiCount === 0 && !nilaiIsEditMode) {
        return showToast('Minimal isi nilai untuk 1 siswa', 'warning');
    }

    // Konfirmasi kalau mode edit
    if (nilaiIsEditMode) {
        const msg = isiNilaiCount === 0
            ? 'Anda akan MENGHAPUS SEMUA nilai untuk slot ini (' + Object.keys(nilaiExistingData).length + ' siswa). Lanjutkan?'
            : 'Update nilai untuk ' + isiNilaiCount + ' siswa? Data lama akan diganti.';
        if (!confirm(msg)) return;
    }

    toggleLoading(true, 'Menyimpan nilai...');
    google.script.run
        .withSuccessHandler(function(r) {
            toggleLoading(false);
            showToast(r);
            document.getElementById('formNilai').style.display = 'none';
            document.getElementById('areaNilaiSiswa').innerHTML = '';
            document.getElementById('nilaiModeEdit').style.display = 'none';
            ['inputNilaiBab','inputNilaiTujuan','inputNilaiBentuk'].forEach(id => document.getElementById(id).value = '');
            nilaiSiswaList = []; nilaiInputRefs = new Map();
            nilaiExistingData = {}; nilaiIsEditMode = false;
        })
        .withFailureHandler(onDataError)
        .simpanNilaiMasal(t, k, mp, kat, bab, tujuan, bentuk, dataSiswa);
};

// === PARSE FILE TEMPLATE (CSV / Excel) ===
function parseImportFromFile(file) {
    const fileName = file.name.toLowerCase();
    const isExcel = fileName.endsWith('.xlsx') || fileName.endsWith('.xls');
    const isCsv = fileName.endsWith('.csv');

    if (!isExcel && !isCsv) {
        return _showImportAlert('danger', 'Format file tidak didukung. Pakai .csv, .xlsx, atau .xls');
    }

    document.getElementById('importFileInfo').innerHTML =
        '<div class="alert alert-info py-2 mb-0 small"><i class="fas fa-spinner fa-spin"></i> Membaca file: <strong>' +
        file.name + '</strong> (' + Math.round(file.size / 1024) + ' KB)...</div>';

    const reader = new FileReader();

    if (isCsv) {
        reader.onload = function(e) {
            const text = e.target.result;
            const rows = _csvTextToRows(text).filter(r => r.some(c => String(c).trim() !== ''));
            document.getElementById('importFileInfo').innerHTML =
                '<div class="alert alert-success py-2 mb-0 small"><i class="fas fa-check"></i> File berhasil dibaca: ' + rows.length + ' baris</div>';
            _processImportRows(rows);
        };
        reader.onerror = function() {
            _showImportAlert('danger', 'Gagal membaca file CSV.');
        };
        reader.readAsText(file, 'UTF-8');
    } else {
        if (typeof XLSX === 'undefined') {
            return _showImportAlert('danger', 'Library Excel belum termuat. Refresh halaman dan coba lagi.');
        }
        reader.onload = function(e) {
            try {
                const data = new Uint8Array(e.target.result);
                const workbook = XLSX.read(data, { type: 'array' });
                const sheetName = workbook.SheetNames[0];
                const sheet = workbook.Sheets[sheetName];
                const rows = XLSX.utils.sheet_to_json(sheet, { header: 1, defval: '', raw: false });
                const cleanRows = rows.filter(r => r.some(c => String(c).trim() !== ''))
                                      .map(r => r.map(c => String(c).trim()));
                document.getElementById('importFileInfo').innerHTML =
                    '<div class="alert alert-success py-2 mb-0 small"><i class="fas fa-check"></i> Sheet "<strong>' + sheetName +
                    '</strong>" berhasil dibaca: ' + cleanRows.length + ' baris</div>';
                _processImportRows(cleanRows);
            } catch (err) {
                _showImportAlert('danger', 'Gagal parse Excel: ' + err.message);
            }
        };
        reader.onerror = function() {
            _showImportAlert('danger', 'Gagal membaca file Excel.');
        };
        reader.readAsArrayBuffer(file);
    }
}

// === HELPER: CSV text → array of arrays (handle quoted comma) ===
function _csvTextToRows(text) {
    if (!text) return [];
    if (text.charCodeAt(0) === 0xFEFF) text = text.substring(1);
    const lines = text.split(/\r?\n/).filter(l => l.length > 0);
    return lines.map(line => {
        const result = [];
        let cur = '', inQuotes = false;
        for (let i = 0; i < line.length; i++) {
            const c = line[i];
            if (c === '"') {
                if (inQuotes && line[i+1] === '"') { cur += '"'; i++; }
                else { inQuotes = !inQuotes; }
            } else if (c === ',' && !inQuotes) {
                result.push(cur.trim()); cur = '';
            } else {
                cur += c;
            }
        }
        result.push(cur.trim());
        return result;
    });
}

function _normHeader(v) {
    return String(v || '').toLowerCase().replace(/[^a-z0-9]/g, '');
}

function _looksLikeHeaderRow(row) {
    const joined = (row || []).map(_normHeader).join('|');
    const hits = IMPORT_TEMPLATE_FIELDS.filter(f => f.aliases.some(a => joined.indexOf(a) >= 0)).length;
    return hits >= 3;
}

function _buildImportHeaderMap(headerRow) {
    const normalized = (headerRow || []).map(_normHeader);
    const map = {};
    IMPORT_TEMPLATE_FIELDS.forEach(field => {
        const idx = normalized.findIndex(h => field.aliases.indexOf(h) >= 0);
        map[field.key] = idx >= 0 ? idx : -1;
    });
    return map;
}

function _rowToImportObject(row, map, idx) {
    const obj = { _idx: idx };
    IMPORT_TEMPLATE_FIELDS.forEach((field, fallbackIdx) => {
        const colIdx = map ? map[field.key] : fallbackIdx;
        obj[field.key] = colIdx >= 0 ? String(row[colIdx] || '').trim() : '';
    });
    return obj;
}

function _processImportRows(rawRows) {
    if (!rawRows || rawRows.length === 0) {
        return _showImportAlert('warning', 'Tidak ada data untuk diproses (mungkin file kosong).');
    }

    const hasHeader = _looksLikeHeaderRow(rawRows[0]);
    const headerMap = hasHeader ? _buildImportHeaderMap(rawRows[0]) : null;
    const dataRows = hasHeader ? rawRows.slice(1) : rawRows;

    importPreviewData = dataRows
        .filter(row => row.some(c => String(c).trim() !== ''))
        .map((row, idx) => _rowToImportObject(row, headerMap, idx));

    if (importPreviewData.length === 0) {
        return _showImportAlert('warning', 'Tidak ada data siswa setelah baris header.');
    }

    _validateAndRender();
}

function _validateAndRender() {
    const existingKelas = new Set(globalDataKelas.map(k => String(k[1]).toLowerCase()));
    const seenSiswa = new Set();

    importPreviewData.forEach((r) => {
        r._err = [];
        r._isNewKelas = false;

        if (!r.kelas) r._err.push('Kelas kosong');
        if (!r.no) r._err.push('No urut kosong');
        if (!r.nama) r._err.push('Nama kosong');

        if (r.kelas) {
            r._isNewKelas = !existingKelas.has(String(r.kelas).toLowerCase());
        }

        const key = String(r.kelas || '').toLowerCase() + '||' + String(r.nama || '').toLowerCase();
        if (r.kelas && r.nama) {
            if (seenSiswa.has(key)) r._err.push('Siswa duplikat di kelas yang sama');
            seenSiswa.add(key);
        }
    });
    _renderImportPreview();
}

function _renderImportPreview() {
    const previewArea = document.getElementById('importPreview');
    previewArea.innerHTML = '';

    if (importPreviewData.length === 0) {
        previewArea.appendChild(el('div', { cls: 'alert alert-warning', text: 'Tidak ada data.' }));
        document.getElementById('btnDoImport').disabled = true;
        document.getElementById('btnDoImportCount').innerText = '';
        return;
    }

    const validCount = importPreviewData.filter(r => r._err.length === 0).length;
    const errorCount = importPreviewData.length - validCount;
    const kelasBaru = new Set(importPreviewData.filter(r => r._err.length === 0 && r._isNewKelas).map(r => r.kelas)).size;

    const summary = el('div', { cls: 'alert ' + (errorCount === 0 ? 'alert-success' : 'alert-warning') });
    summary.appendChild(el('strong', { text: validCount + ' siswa valid' }));

function doImportSiswa() {
    const validData = importPreviewData
        .filter(r => r._err.length === 0)
        .map(r => {
            const obj = {};
            IMPORT_TEMPLATE_FIELDS.forEach(f => obj[f.key] = r[f.key] || '');
            return obj;
        });

    if (validData.length === 0) return showToast('Tidak ada data valid untuk diimpor', 'warning');

    const errorCount = importPreviewData.length - validData.length;
    const kelasBaru = new Set(importPreviewData.filter(r => r._err.length === 0 && r._isNewKelas).map(r => r.kelas)).size;
    let confirmMsg = 'Import ' + validData.length + ' siswa';
    if (kelasBaru > 0) confirmMsg += ' dan buat ' + kelasBaru + ' kelas baru';
    confirmMsg += '?';
    if (errorCount > 0) confirmMsg += '\n(' + errorCount + ' baris dengan error akan di-skip)';
    if (!confirm(confirmMsg)) return;

    toggleLoading(true, 'Mengimpor kelas & siswa...');
    google.script.run
        .withSuccessHandler(function(res) {
            toggleLoading(false);
            bootstrap.Modal.getInstance(document.getElementById('modalImport')).hide();
            applyFastMutation(res);
            showToast(res.pesan, res.gagal > 0 ? 'warning' : 'success');
            if (res.errors && res.errors.length > 0) {
                console.warn('Import errors dari server:', res.errors);
            }
        })
        .withFailureHandler(onDataError)
        .importSiswaMassalCepat(validData, _actor());
}

function _triggerDownload(blob, filename) {
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url; a.download = filename;
    document.body.appendChild(a); a.click();
    setTimeout(() => { document.body.removeChild(a); URL.revokeObjectURL(url); }, 100);
}

function getRekapParams() {
    const mode = document.querySelector('input[name="modeRekap"]:checked').value;
    const kelas = document.getElementById('inputRekapKelas').value;
    const mapel = document.getElementById('inputRekapMapel').value;
    const jenis = document.getElementById('inputRekapJenis').value;
    let tglMulai = '', tglAkhir = '', bulan = null, tahun = null;
    if (mode === 'bulanan') {
        bulan = document.getElementById('inputRekapBulan').value;
        tahun = document.getElementById('inputRekapTahun').value;
    } else {
        tglMulai = document.getElementById('inputRekapMulai').value;
        tglAkhir = document.getElementById('inputRekapAkhir').value;
    }
    return { mode, kelas, mapel, jenis, tglMulai, tglAkhir, bulan, tahun };
}

function validateRekapParams(p) {
    if (!p.jenis) { showToast('Pilih jenis data!', 'warning'); return false; }
    if (!p.mapel) { showToast('Pilih mata pelajaran!', 'warning'); return false; }
    if (p.mode === 'custom' && (!p.tglMulai || !p.tglAkhir)) {
        showToast('Lengkapi rentang tanggal!', 'warning'); return false;
    }
    if (p.mode === 'custom' && p.tglMulai > p.tglAkhir) {
        showToast('Tanggal mulai harus sebelum tanggal akhir!', 'warning'); return false;
    }
    if (p.jenis === 'Absensi' && (p.kelas === 'Semua' || !p.kelas)) {
        showToast('Untuk Rekap Absensi, pilih kelas spesifik (bukan Semua Kelas).', 'warning'); return false;
    }
    return true;
}

function toggleModeRekap() {
    const mode = document.querySelector('input[name="modeRekap"]:checked').value;
    document.getElementById('areaModeBulanan').style.display = (mode === 'bulanan') ? 'block' : 'none';
    document.getElementById('areaModeCustom').style.display = (mode === 'custom') ? 'block' : 'none';
}

function onJenisRekapChange() {
    const jenis = document.getElementById('inputRekapJenis').value;
    const hint = document.getElementById('hintKelasRekap');
    const kelasSel = document.getElementById('inputRekapKelas');
    if (jenis === 'Absensi') {
        hint.innerText = 'Rekap Absensi memerlukan kelas spesifik.';
        // Otomatis pindah dari "Semua" kalau jenisnya Absensi
        if (kelasSel.value === 'Semua' && kelasSel.options.length > 1) {
            kelasSel.value = kelasSel.options[1].value;
        }
    } else {
        hint.innerText = '';
    }
}

function initRekapTahun() {
    const sel = document.getElementById('inputRekapTahun');
    if (!sel || sel.options.length > 0) return;
    const now = new Date();
    const thisYear = now.getFullYear();
    for (let y = thisYear + 1; y >= thisYear - 3; y--) {
        const opt = el('option', { text: y, attrs: { value: y } });
        if (y === thisYear) opt.selected = true;
        sel.appendChild(opt);
    }
    // Default bulan = bulan ini
    document.getElementById('inputRekapBulan').value = (now.getMonth() + 1);
}

function sameRekapParams(a, b) {
    if (!a || !b) return false;
    return a.mode === b.mode && a.kelas === b.kelas && a.mapel === b.mapel && a.jenis === b.jenis &&
        a.tglMulai === b.tglMulai && a.tglAkhir === b.tglAkhir &&
        String(a.bulan) === String(b.bulan) && String(a.tahun) === String(b.tahun);
}

function getPreviewForOutput(p, callback, loadingText) {
    if (lastPreviewRekapData && sameRekapParams(lastPreviewRekapParams, p)) {
        callback(lastPreviewRekapData);
        return;
    }
    toggleLoading(true, loadingText || 'Mengambil data rekap...');
    google.script.run
        .withSuccessHandler(function(res) {
            toggleLoading(false);
            lastPreviewRekapData = res;
            lastPreviewRekapParams = p;
            callback(res);
        })
        .withFailureHandler(onDataError)
        .getPreviewRekap(p.kelas, p.jenis, p.tglMulai, p.tglAkhir, p.mode, p.bulan, p.tahun, p.mapel);
}

function escHtml(v) {
    return String(v == null ? '' : v)
        .replace(/&/g, '&amp;')
        .replace(/</g, '&lt;')
        .replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;')
        .replace(/'/g, '&#039;');
}

function safeFileName(v) {
    return String(v || 'rekap').replace(/[^a-z0-9_\-]+/gi, '_').replace(/_+/g, '_');
}

function csvEscape(v) {
    const s = String(v == null ? '' : v);
    return /[",\n]/.test(s) ? '"' + s.replace(/"/g, '""') + '"' : s;
}

function buildCsvFromPreview(res, p) {
    const periode = res.periodeLabel || '';
    const lines = [];
    lines.push(['REKAPITULASI ' + p.jenis]);
    lines.push(['Kelas', p.kelas]);
    lines.push(['Mata Pelajaran', p.mapel || (res.mapel || '-')]);
    lines.push(['Periode', periode]);
    lines.push([]);
    if (res.isPivot || p.jenis === 'Absensi') {
        const header = ['No', 'Nama Siswa'].concat(res.headerTanggal || []).concat(['H','S','I','A','T']);
        lines.push(header);
        (res.rows || []).forEach(r => lines.push([r.no, r.nama].concat(r.cells || []).concat([r.H, r.S, r.I, r.A, r.T])));
    } else if (res.isGrouped && p.jenis === 'Nilai') {
        lines.push(['Tanggal','Mata Pelajaran','Kategori','Bab','Bentuk','Tujuan Pembelajaran','No','Nama Siswa','Nilai']);
        (res.data || []).forEach(g => (g.siswa || []).forEach(s => lines.push([g.tanggal, g.mapel, g.kategori, g.bab, g.bentuk, g.tujuan, s[0], s[1], s[2]])));
    } else {
        lines.push(res.headers || []);
        (res.data || []).forEach(row => lines.push(row));
    }
    return '\uFEFF' + lines.map(row => row.map(csvEscape).join(',')).join('\n');
}

function downloadCsvBlob(csv, filename) {
    const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = filename;
    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);
    setTimeout(() => URL.revokeObjectURL(url), 1000);
}

function renderKelasCheckboxes(selected) {

function submitPengampu() {
    const id = document.getElementById('pengampuId').value.trim();
    const data = {
        nama: document.getElementById('pengampuNama').value.trim(),
        nip: document.getElementById('pengampuNip').value.trim(),
        noHp: document.getElementById('pengampuNoHp').value.trim(),
        username: document.getElementById('pengampuUsername').value.trim(),
        password: document.getElementById('pengampuPassword').value.trim(),
        kelasDiampu: Array.from(document.querySelectorAll('.pengampu-kelas-chk:checked')).map(c => c.value),
        active: document.getElementById('pengampuActive').value !== 'Tidak'
    };
    if (!data.nama || !data.username) return showToast('Nama & username wajib diisi', 'error');
    if (!id && (!data.password || data.password.length < 4)) return showToast('Password minimal 4 karakter', 'error');
    const cb = function(res) {
        showToast(res.message, 'success');
        pengampuCache = res.pengampu || [];
        renderPengampuTable();
        bootstrap.Modal.getInstance(document.getElementById('modalPengampu')).hide();
    };
    if (id) google.script.run.withSuccessHandler(cb).withFailureHandler(handleError).editPengampuCepat(id, data);
    else google.script.run.withSuccessHandler(cb).withFailureHandler(handleError).simpanPengampuCepat(data);
}

let _checkinTimer = null;
function startCheckinClock() {
    const tick = () => {
        const now = new Date();
        const pad = n => String(n).padStart(2, '0');
        const el = document.getElementById('checkinClock');
        const dt = document.getElementById('checkinDate');
        if (el) el.innerText = pad(now.getHours())+':'+pad(now.getMinutes())+':'+pad(now.getSeconds());
        if (dt) {
            const hariArr = ['Minggu','Senin','Selasa','Rabu','Kamis','Jumat','Sabtu'];
            const bulanArr = ['Januari','Februari','Maret','April','Mei','Juni','Juli','Agustus','September','Oktober','November','Desember'];
            dt.innerText = hariArr[now.getDay()]+', '+now.getDate()+' '+bulanArr[now.getMonth()]+' '+now.getFullYear();
        }
    };
    tick();
    if (_checkinTimer) return;
    _checkinTimer = setInterval(tick, 1000);
    loadCheckinStatus();
    loadAbsensiPengampu();
}

function currentWeekStr(){
    const d = new Date();
    d.setHours(0,0,0,0);
    d.setDate(d.getDate() + 4 - (d.getDay()||7));
    const y = d.getFullYear();
    const w = Math.ceil((((d - new Date(y,0,1))/86400000) + 1)/7);
    return y + '-W' + String(w).padStart(2,'0');
}

// Format 'yyyy-mm-dd' -> 'Rabu, 06/08/2026'
function formatTanggalId(s){
    if(!s) return '';
    var parts = String(s).split('-');
    if(parts.length !== 3) return s;
    var d = new Date(parseInt(parts[0]), parseInt(parts[1])-1, parseInt(parts[2]));
    if(isNaN(d.getTime())) return s;
    var hari = ['Minggu','Senin','Selasa','Rabu','Kamis','Jumat','Sabtu'];
    var p = function(n){ return String(n).padStart(2,'0'); };
    return hari[d.getDay()] + ', ' + p(d.getDate()) + '/' + p(d.getMonth()+1) + '/' + d.getFullYear();
}

function handleLogoUpload(evt) {
    const file = evt.target.files && evt.target.files[0];
    if (!file) return;
    if (file.size > 500 * 1024) {
        if (!confirm('Ukuran file cukup besar (' + Math.round(file.size/1024) + ' KB). Google Sheets punya limit ~50KB per cell. Tetap lanjut?')) return;
    }
    const reader = new FileReader();
    reader.onload = function(e) {
        const dataUrl = e.target.result;
        document.getElementById('inputSetLogo').value = dataUrl;
        applyLogo(dataUrl);
        showToast('Logo dimuat. Klik Simpan untuk menyimpan permanen.', 'success');
    };
    reader.readAsDataURL(file);
}

    const monday = new Date(d.getTime() - (dow - 1) * 86400000);

    const fmt = x => x.getFullYear()+'-'+pad(x.getMonth()+1)+'-'+pad(x.getDate());

function renderKehadiranChart(stats){
    if (!stats || typeof Chart === 'undefined') return;
    const canvas = document.getElementById('chartKehadiran');
    if (!canvas) return;
    const keys = stats.keys || ['H','S','I','A','T'];
    const isPct = _dashMode === 'pct';
    const dataVals = keys.map(k => isPct ? (stats.pct[k]||0) : (stats.counts[k]||0));
    const colors = ['#198754','#ffc107','#0dcaf0','#dc3545','#6610f2'];

    const summary = document.getElementById('dashKehadiranSummary');
    if (summary) {
        summary.innerHTML = 'Cakupan: <b>'+escHtml(stats.kelas||'Semua')+'</b> · Periode <b>'+escHtml(stats.tglMulai)+'</b> s.d. <b>'+escHtml(stats.tglAkhir)+'</b> — Total entri: <b>'+stats.total+'</b> | '+
            stats.labels.map(function(l,idx){ return escHtml(l)+': '+stats.counts[keys[idx]]+' ('+stats.pct[keys[idx]]+'%)'; }).join(' · ');
    }

    // Bentuk bagan berubah -> bangun ulang chart
    if (_kehadiranChart && _kehadiranChartType !== _dashChartType) {
        _kehadiranChart.destroy();
        _kehadiranChart = null;
    }

    const unit = isPct ? '%' : ' siswa';
    if (_kehadiranChart) {
        _kehadiranChart.data.labels = stats.labels;
        _kehadiranChart.data.datasets[0].data = dataVals;
        _kehadiranChart.update();
        return;
    }
    _kehadiranChartType = _dashChartType;
    if (_dashChartType === 'pie') {
        _kehadiranChart = new Chart(canvas.getContext('2d'), {
            type: 'pie',
            data: { labels: stats.labels, datasets: [{ data: dataVals, backgroundColor: colors, borderColor: 'rgba(255,255,255,.15)', borderWidth: 2 }] },
            options: {
                responsive: true, maintainAspectRatio: false,
                plugins: {
                    legend: { display: true, position: 'bottom', labels: { color: '#c7cdea', padding: 14 } },
                    tooltip: { callbacks: { label: function(c){ return ' ' + c.label + ': ' + c.parsed + unit; } } }
                }
            }
        });
    } else {
        _kehadiranChart = new Chart(canvas.getContext('2d'), {
            type: 'bar',
            data: { labels: stats.labels, datasets: [{ label: isPct ? 'Persentase (%)' : 'Jumlah Siswa', data: dataVals, backgroundColor: colors, borderRadius: 6 }] },
            options: {
                responsive: true, maintainAspectRatio: false,
                plugins: {
                    legend: { display: false },
                    tooltip: { callbacks: { label: function(c){ return c.parsed.y + unit; } } }
                },
                scales: { y: { beginAtZero: true, max: isPct ? 100 : undefined, ticks: { callback: function(v){ return isPct ? v+'%' : v; } } } }
            }
        });
    }
}

function downloadExcelBukuMingguan(){

    toggleLoading(true, 'Menyiapkan CSV...');
    google.script.run.withSuccessHandler(function(data){
        toggleLoading(false);
        data = data || [];
        let kumpul=0, tdk=0;
        data.forEach(function(d){ if(d.status==='Kumpul')kumpul++; else if(d.status==='Tidak Kumpul')tdk++; });
        const lines = [];
        lines.push(['REKAP ABSEN BUKU MINGGUAN']);

        lines.push(['Jenis Ibadah', jenisIbadah]);

  // ---------- CHECKBOX RENDERERS ----------
  function chkKelas(containerId, selected, onlyList){
    var el=document.getElementById(containerId); if(!el) return;
    var list=onlyList||kelasNames();
    var sel={}; (selected||[]).forEach(function(k){sel[k]=true;});
    if(!list.length){ el.innerHTML='<span class="text-muted small">Belum ada kelas.</span>'; return; }
    el.innerHTML=list.map(function(k){ return '<label class="fb-chip"><input type="checkbox" class="'+containerId+'-chk" value="'+a(k)+'" '+(sel[k]?'checked':'')+'> '+h(k)+'</label>'; }).join('');
  }
  function chkGroups(containerId, selected){
    var el=document.getElementById(containerId); if(!el) return;
    var sel={}; (selected||[]).forEach(function(k){sel[k]=true;});
    if(!window._fbGroups.length){ el.innerHTML='<span class="text-muted small">Belum ada kelompok. Buat di menu Kelompok Kelas.</span>'; return; }
    el.innerHTML=window._fbGroups.map(function(g){ return '<label class="fb-chip"><input type="checkbox" class="'+containerId+'-chk" value="'+a(g.id)+'" '+(sel[g.id]?'checked':'')+'> '+h(g.nama)+'</label>'; }).join('');
  }
  function getChecked(cls){ return Array.prototype.slice.call(document.querySelectorAll('.'+cls+'-chk:checked')).map(function(c){return c.value;}); }

  window.fbUploadBahanAjar=function(){

    var file=fi&&fi.files&&fi.files[0];
    if(!judul) return toast('Judul wajib diisi','warning');
    if(!file) return toast('Pilih file PDF dulu','warning');
    if(file.type!=='application/pdf' && !/\.pdf$/i.test(file.name)) return toast('Hanya file PDF yang diperbolehkan','warning');
    var st=document.getElementById('baUploadStatus'); if(st) st.textContent='Membaca file...';
    var reader=new FileReader();
    reader.onload=function(){
      var base64=String(reader.result||'').split(',')[1]||'';
      if(st) st.textContent='Mengunggah ke Google Drive...';
      var data={judul:judul, deskripsi:document.getElementById('baDeskripsi').value.trim(), kelompokId:document.getElementById('baKelompok').value, namaFile:file.name, mimeType:file.type||'application/pdf', base64:base64};
      run().withSuccessHandler(function(m){ if(st) st.textContent=''; toast(m,'success'); document.getElementById('baJudul').value=''; document.getElementById('baDeskripsi').value=''; fi.value=''; fbLoadBahanAjar(); }).withFailureHandler(function(e){ if(st) st.textContent=''; err(e); }).uploadBahanAjar(ctx(), data);
    };
    reader.onerror=function(){ if(st) st.textContent=''; toast('Gagal membaca file','error'); };
    reader.readAsDataURL(file);
